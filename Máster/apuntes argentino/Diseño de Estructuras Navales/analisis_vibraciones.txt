%% SCRIPT DE CÁLCULO DE VIBRACIONES (TIPO PÓRTICO/VIGA)
%  Genera Matrices K y M Globales y calcula Frecuencias Propias.
%  Basado en Matriz de Masa Consistente (Flexión Compuesta).

clc; clear; close all;
fprintf('--- INICIANDO CÁLCULO DE ESTRUCTURA NAVAL/CIVIL ---\n');

%% 1. DEFINICIÓN DE PARÁMETROS (MODIFICAR AQUÍ)

% --- MATERIALES (Ejemplo: Acero) ---
E_mat   = 210e9;       % Módulo de Young (Pa) -> 210 GPa
rho_mat = 7850;        % Densidad (kg/m^3)

% --- SECCIONES (Ejemplo genérico, pon los valores reales de tu perfil) ---
% Supongamos un perfil para vigas y otro para columnas, o el mismo.
A_sec = 0.005;    % Área (m^2)
I_sec = 8e-5;     % Inercia (m^4)

% --- GEOMETRÍA (NODOS) ---
% Basado en tu Ejercicio 10. Origen (0,0) en el apoyo izquierdo.
% A=(0,2), B=(2,2), ApoyoB=(2,0), C=(5,2), D=(5,4), E=(7,2), F=(5,-1)
% Formato: [ID_Nodo, Coord_X, Coord_Y]
Nodes = [
    1,  0.0,  2.0;   % A (Extremo izquierdo)
    2,  2.0,  2.0;   % B (Unión T izquierda)
    3,  2.0,  0.0;   % Apoyo bajo B (Articulación fija al suelo)
    4,  5.0,  2.0;   % C (Unión Cruz)
    5,  5.0,  4.0;   % D (Extremo superior)
    6,  5.0, -1.0;   % F (Apoyo bajo C, a 3m de altura desde C) -> C(y=2) - 3m = -1
    7,  7.0,  2.0;   % E (Extremo derecho)
];

% --- CONECTIVIDAD (ELEMENTOS) ---
% Define cómo se unen los nodos.
% Formato: [ID_Elem, Nodo_i, Nodo_j, E, Area, Inercia, Densidad]
Elements = [
    % Viga Horizontal
    1,  1, 2,  E_mat, A_sec, I_sec, rho_mat;  % Tramo A-B (2m)
    2,  2, 4,  E_mat, A_sec, I_sec, rho_mat;  % Tramo B-C (3m)
    3,  4, 7,  E_mat, A_sec, I_sec, rho_mat;  % Tramo C-E (2m)
    
    % Elementos Verticales
    4,  3, 2,  E_mat, A_sec, I_sec, rho_mat;  % Pilar bajo B (2m)
    5,  6, 4,  E_mat, A_sec, I_sec, rho_mat;  % Pilar bajo C (3m)
    6,  4, 5,  E_mat, A_sec, I_sec, rho_mat;  % Pilar superior C-D (2m)
];

% --- CONDICIONES DE CONTORNO (APOYOS) ---
% Formato: [ID_Nodo, Restricción_X, Restricción_Y, Restricción_Giro]
% 1 = Fijo (Impedido), 0 = Libre
% En el dibujo, los apoyos en el suelo (nodos 3 y 6) son articulados (triángulos).
BCs = [
    3, 1, 1, 0;  % Apoyo en el suelo bajo B (Fijo X, Fijo Y, Giro Libre)
    6, 1, 1, 0;  % Apoyo en el suelo F (Fijo X, Fijo Y, Giro Libre)
];

%% 2. ENSAMBLAJE DE MATRICES (NÚCLEO DE CÁLCULO)

num_nodes = size(Nodes, 1);
num_dofs  = num_nodes * 3; % 3 Grados de libertad por nodo (u, v, theta)

% Inicializamos matrices globales vacías
K_Global = zeros(num_dofs, num_dofs);
M_Global = zeros(num_dofs, num_dofs);

fprintf('Ensamblando matriz de rigidez y masa consistente...\n');

for i = 1:size(Elements, 1)
    % 2.1 Extraer datos del elemento
    n1 = Elements(i, 2);
    n2 = Elements(i, 3);
    E_e = Elements(i, 4); A_e = Elements(i, 5); I_e = Elements(i, 6); rho_e = Elements(i, 7);
    
    % 2.2 Geometría del elemento
    x1 = Nodes(n1, 2); y1 = Nodes(n1, 3);
    x2 = Nodes(n2, 2); y2 = Nodes(n2, 3);
    Lx = x2 - x1;
    Ly = y2 - y1;
    L  = sqrt(Lx^2 + Ly^2);
    
    c = Lx / L; % Coseno director
    s = Ly / L; % Seno director
    
    % 2.3 Matriz de Rigidez Local (k) - Pórtico Plano
    k_local = [
         E_e*A_e/L,   0,             0,              -E_e*A_e/L,  0,             0;
         0,           12*E_e*I_e/L^3, 6*E_e*I_e/L^2,  0,          -12*E_e*I_e/L^3, 6*E_e*I_e/L^2;
         0,           6*E_e*I_e/L^2,  4*E_e*I_e/L,    0,          -6*E_e*I_e/L^2,  2*E_e*I_e/L;
        -E_e*A_e/L,   0,             0,               E_e*A_e/L,  0,             0;
         0,          -12*E_e*I_e/L^3,-6*E_e*I_e/L^2,  0,           12*E_e*I_e/L^3, -6*E_e*I_e/L^2;
         0,           6*E_e*I_e/L^2,  2*E_e*I_e/L,    0,          -6*E_e*I_e/L^2,  4*E_e*I_e/L
    ];
    
    % 2.4 Matriz de Masa Consistente Local (m) - Flexión Compuesta
    % Factor común: rho * A * L / 420
    cte = (rho_e * A_e * L) / 420;
    
    % Matriz simétrica completa (u1, v1, th1, u2, v2, th2)
    m_local = cte * [
         140,  0,      0,        70,   0,      0;
         0,    156,    22*L,     0,    54,     -13*L;
         0,    22*L,   4*L^2,    0,    13*L,   -3*L^2;
         70,   0,      0,        140,  0,      0;
         0,    54,     13*L,     0,    156,    -22*L;
         0,    -13*L,  -3*L^2,   0,    -22*L,  4*L^2
    ];
    
    % 2.5 Matriz de Transformación (T)
    T = zeros(6,6);
    bloque = [c s 0; -s c 0; 0 0 1];
    T(1:3, 1:3) = bloque;
    T(4:6, 4:6) = bloque;
    
    % 2.6 Rotación a Globales
    k_glob_elem = T' * k_local * T;
    m_glob_elem = T' * m_local * T;
    
    % 2.7 Ensamblaje (Mapping de índices)
    % Grados de libertad globales: [3*n1-2, 3*n1-1, 3*n1, ...]
    idx = [3*n1-2, 3*n1-1, 3*n1, 3*n2-2, 3*n2-1, 3*n2];
    
    K_Global(idx, idx) = K_Global(idx, idx) + k_glob_elem;
    M_Global(idx, idx) = M_Global(idx, idx) + m_glob_elem;
end

%% 3. APLICACIÓN DE CONDICIONES DE CONTORNO

fixed_dofs = [];
for k = 1:size(BCs, 1)
    nodo_bc = BCs(k, 1);
    % Verificar X
    if BCs(k, 2) == 1, fixed_dofs = [fixed_dofs, 3*nodo_bc-2]; end
    % Verificar Y
    if BCs(k, 3) == 1, fixed_dofs = [fixed_dofs, 3*nodo_bc-1]; end
    % Verificar Giro
    if BCs(k, 4) == 1, fixed_dofs = [fixed_dofs, 3*nodo_bc];   end
end

% Grados de libertad libres (los que sí se mueven)
free_dofs = setdiff(1:num_dofs, fixed_dofs);

% Reducción de matrices (eliminamos filas/cols de apoyos)
K_red = K_Global(free_dofs, free_dofs);
M_red = M_Global(free_dofs, free_dofs);

%% 4. SOLUCIÓN DEL PROBLEMA DE AUTOVALORES

% Resolvemos [K]{phi} = w^2 [M]{phi}
[V_red, D] = eig(K_red, M_red);

% Extraer diagonal de autovalores (omega cuadrado)
omega_sq = diag(D);

% Ordenar de menor a mayor
[omega_sq, ind] = sort(omega_sq);
V_red = V_red(:, ind);

% Calcular frecuencias
omega = sqrt(abs(omega_sq));  % Rad/s
f_hz  = omega / (2*pi);       % Hz

%% 5. RESULTADOS Y VISUALIZACIÓN

fprintf('\n--- RESULTADOS OBTENIDOS ---\n');
fprintf('Las matrices K_Global y M_Global están disponibles en el Workspace.\n');
fprintf('Primeras 5 frecuencias naturales:\n');
fprintf('Modo \t Frecuencia (Hz) \t Omega (rad/s)\n');
for k = 1:min(5, length(f_hz))
    fprintf('%d \t\t %.4f \t\t\t %.4f\n', k, f_hz(k), omega(k));
end

% PLOT BÁSICO DE LA ESTRUCTURA (Para verificar geometría)
figure(1); clf; hold on; grid on; axis equal;
title('Geometría de la Estructura (Nodos y Elementos)');
xlabel('X (m)'); ylabel('Y (m)');

% Dibujar elementos
for i = 1:size(Elements, 1)
    n1 = Elements(i, 2);
    n2 = Elements(i, 3);
    x_line = [Nodes(n1, 2), Nodes(n2, 2)];
    y_line = [Nodes(n1, 3), Nodes(n2, 3)];
    plot(x_line, y_line, 'b-o', 'LineWidth', 2, 'MarkerFaceColor', 'r');
    % Etiquetar número de elemento en el centro
    text(mean(x_line), mean(y_line), sprintf(' E%d', Elements(i,1)), 'Color', 'k');
end

% Etiquetar nodos
for i = 1:size(Nodes, 1)
    text(Nodes(i, 2)+0.1, Nodes(i, 3)+0.1, sprintf('N%d', Nodes(i,1)), ...
        'FontWeight', 'bold', 'Color', 'r');
end

% Dibujar apoyos (Simbolizados con triángulos o marcas)
plot(Nodes(BCs(:,1), 2), Nodes(BCs(:,1), 3), 'k^', 'MarkerSize', 10, 'MarkerFaceColor', 'k');

disp(' ');
disp('Proceso finalizado. Revisa las variables en el Workspace.');
